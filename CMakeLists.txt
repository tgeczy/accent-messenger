cmake_minimum_required(VERSION 3.21)
project(messenger LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(UNICORN_SOURCE "${CMAKE_SOURCE_DIR}/.build/unicorn-2.1.4/src" CACHE PATH "Unmodified Unicorn 2.1.4 source")
set(YY_THUNKS_ROOT "${CMAKE_SOURCE_DIR}/native/third_party/YY-Thunks" CACHE PATH "YY-Thunks object distribution")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(UNICORN_ARCH x86 CACHE STRING "" FORCE)
set(UNICORN_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(UNICORN_INSTALL OFF CACHE BOOL "" FORCE)
set(UNICORN_LEGACY_STATIC_ARCHIVE OFF CACHE BOOL "" FORCE)
add_compile_definitions(WINVER=0x0601 _WIN32_WINNT=0x0601 NOMINMAX)
add_subdirectory("${UNICORN_SOURCE}" unicorn)
add_library(messenger SHARED native/src/api.cpp native/src/frontend.cpp native/src/renderer.cpp native/src/noise.cpp)
target_include_directories(messenger PRIVATE native/include "${UNICORN_SOURCE}/include")
target_link_libraries(messenger PRIVATE unicorn)
if(MSVC)
    target_compile_options(messenger PRIVATE /W4 /fp:strict /utf-8)
    if(CMAKE_SIZEOF_VOID_P EQUAL 8)
        set(ARCH x64)
    else()
        set(ARCH x86)
    endif()
    set(THUNKS "${YY_THUNKS_ROOT}/objs/${ARCH}/YY_Thunks_for_Win7.obj")
    if(NOT EXISTS "${THUNKS}")
        message(FATAL_ERROR "Missing Windows 7 thunks: ${THUNKS}")
    endif()
    target_sources(messenger PRIVATE "${THUNKS}")
    target_link_options(messenger PRIVATE /SUBSYSTEM:WINDOWS,6.01 /OPT:REF /OPT:ICF)
endif()

option(MESSENGER_BUILD_SAPI "Build the SAPI engine and native settings dialog" OFF)
if(MESSENGER_BUILD_SAPI)
    enable_language(RC)
    add_subdirectory(sapi)
endif()
