set(SAPI_COMMON settings.cpp text.cpp)
add_library(messenger_sapi SHARED engine.cpp runtime.cpp ${SAPI_COMMON} messenger_sapi.def
    ../native/src/api.cpp ../native/src/frontend.cpp ../native/src/renderer.cpp ../native/src/noise.cpp)
target_include_directories(messenger_sapi PRIVATE ../native/include "${UNICORN_SOURCE}/include" "${CMAKE_SOURCE_DIR}/.build")
target_link_libraries(messenger_sapi PRIVATE unicorn sapi ole32 advapi32 shell32 bcrypt normaliz)
add_executable(messenger_settings WIN32 settings_gui.cpp settings.cpp settings.rc)
target_link_libraries(messenger_settings PRIVATE sapi ole32 shell32 user32 advapi32)
add_executable(sapi_test test.cpp ${SAPI_COMMON})
target_link_libraries(sapi_test PRIVATE sapi ole32 shell32 advapi32 normaliz)
add_executable(settings_test settings_test.cpp settings.cpp settings.rc)
target_link_libraries(settings_test PRIVATE sapi ole32 shell32 user32 advapi32)
foreach(target messenger_sapi messenger_settings sapi_test settings_test)
    target_compile_definitions(${target} PRIVATE UNICODE _UNICODE)
    target_compile_options(${target} PRIVATE /W4 /fp:strict /utf-8)
    target_sources(${target} PRIVATE "${THUNKS}")
    target_link_options(${target} PRIVATE /OPT:REF /OPT:ICF)
endforeach()
target_link_options(messenger_sapi PRIVATE /SUBSYSTEM:WINDOWS,6.01)
target_link_options(messenger_settings PRIVATE /SUBSYSTEM:WINDOWS,6.01)
target_link_options(sapi_test PRIVATE /SUBSYSTEM:CONSOLE,6.01)
target_link_options(settings_test PRIVATE /SUBSYSTEM:CONSOLE,6.01)
